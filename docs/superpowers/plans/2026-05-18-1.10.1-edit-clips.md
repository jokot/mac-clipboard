> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship MaClip 1.10.1 — right-click a text / URL / file clip → "Edit…" → inline `TextEditor` with Cancel + Save. In-place update preserves id/date/source/concealed/OCR flags. Tag cache invalidates so the row re-detects after edit.

**Architecture:** New `replaceContent(of:with:)` method on `ClipboardListViewModel`. `ClipboardItemRow` gains `@State isEditing/editText/editError`, an `EditorView` inline subview, an `Edit…` menu entry in the unified context menu, and a commit handler that parses content kind appropriately. `ContentView` wires an `onEdit` closure.

**Tech Stack:** Swift, SwiftUI (`TextEditor`, `ScrollView`, `Button.keyboardShortcut(.cancelAction/.defaultAction)`), XCTest. Build: `xcodebuild -scheme MacClipboard -destination 'platform=macOS' test`.

```swift
@discardableResult
func replaceContent(of item: ClipboardItem, with newContent: ClipboardItemContent) -> ClipboardItem? {
    guard let idx = items.firstIndex(where: { $0.id == item.id }) else { return nil }
    let existing = items[idx]
    let updated = ClipboardItem(
        id: existing.id,
        date: existing.date,
        content: newContent,
        sourceBundleID: existing.sourceBundleID,
        isConcealed: existing.isConcealed,
        concealedExpiresAt: existing.concealedExpiresAt,
        isOCRResult: existing.isOCRResult
    )
    items[idx] = updated
    repository.saveToDiskAsync(items: items)
    ContentTagDetector.clearCache()
    return updated
}
```

If `ClipboardItem`'s initializer signature doesn't match exactly (param names/order), adapt to whatever the existing init uses — keep id/date/sourceBundleID/isConcealed/concealedExpiresAt/isOCRResult preserved; only `content` changes.

- [ ] **Step 2: Add edit helpers**

Add as private methods on `ClipboardItemRow`:

```swift
private func canEdit(_ content: ClipboardItemContent) -> Bool {
    switch content {
    case .text, .url, .file: return true
    case .image:             return false
    }
}

private func enterEditMode() {
    editText = currentContentAsText(item.content)
    editError = nil
    isEditing = true
}

private func currentContentAsText(_ content: ClipboardItemContent) -> String {
    switch content {
    case .text(let s):  return s
    case .url(let u):   return u.absoluteString
    case .file(let us): return us.map(\.path).joined(separator: "\n")
    case .image:        return ""
    }
}

private func commitEdit() {
    switch item.content {
    case .text:
        onEdit?(item, .text(editText))
        isEditing = false
    case .url:
        let trimmed = editText.trimmingCharacters(in: .whitespacesAndNewlines)
        guard let u = URL(string: trimmed) else {
            editError = "Not a valid URL"
            return
        }
        onEdit?(item, .url(u))
        isEditing = false
    case .file:
        let lines = editText.split(separator: "\n", omittingEmptySubsequences: true)
        let urls = lines.map { URL(fileURLWithPath: String($0).trimmingCharacters(in: .whitespacesAndNewlines)) }
        guard !urls.isEmpty else {
            editError = "Need at least one path"
            return
        }
        onEdit?(item, .file(urls))
        isEditing = false
    case .image:
        isEditing = false
    }
}
```

```swift
private struct EditorView: View {
    @Binding var text: String
    @Binding var error: String?
    let onCancel: () -> Void
    let onSave: () -> Void

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            ScrollView {
                TextEditor(text: $text)
                    .font(.body)
                    .frame(minHeight: 80)
            }
            .frame(maxHeight: 240)
            .overlay(
                RoundedRectangle(cornerRadius: 6)
                    .stroke(Color.secondary.opacity(0.3), lineWidth: 1)
            )
            if let error = error {
                Text(error)
                    .font(.caption)
                    .foregroundColor(.red)
            }
            HStack {
                Spacer()
                Button("Cancel", action: onCancel)
                    .keyboardShortcut(.cancelAction)
                Button("Save", action: onSave)
                    .keyboardShortcut(.defaultAction)
            }
        }
    }
}
```
